\section*{APPENDIX A: API DOCUMENTATION}
\addcontentsline{toc}{section}{Appendix A: API Documentation}

\subsection*{Authentication Endpoints}
\begin{itemize}
\item \textbf{POST /api/auth/login} - User login
\item \textbf{POST /api/auth/register} - User registration  
\item \textbf{GET /api/auth/google} - Google OAuth
\item \textbf{POST /api/auth/logout} - User logout
\end{itemize}

\subsection*{Product Management Endpoints}
\begin{itemize}
\item \textbf{GET /api/products} - Retrieve products with pagination
\item \textbf{POST /api/products} - Create new product
\item \textbf{PUT /api/products/:id} - Update product
\item \textbf{DELETE /api/products/:id} - Delete product
\item \textbf{GET /api/products/dashboard-stats} - Analytics
\end{itemize}

\subsection*{Sales Management Endpoints}
\begin{itemize}
\item \textbf{GET /api/sales} - Retrieve sales records
\item \textbf{POST /api/sales} - Create new sale
\item \textbf{PUT /api/sales/:id} - Update sale
\item \textbf{DELETE /api/sales/:id} - Cancel sale
\end{itemize}

\section*{APPENDIX B: DATABASE SCHEMAS}
\addcontentsline{toc}{section}{Appendix B: Database Schemas}

\subsection*{Product Schema}
\begin{lstlisting}[caption=Product Schema]
{
  name: String (required),
  sku: String (required, unique),
  description: String,
  category: ObjectId (ref: Category),
  retailPrice: Number (required),
  wholesalePrice: Number,
  stockQuantity: Number (default: 0),
  createdBy: ObjectId (ref: User),
  isActive: Boolean (default: true)
}
\end{lstlisting}

\subsection*{Sale Schema}
\begin{lstlisting}[caption=Sale Schema]
{
  saleId: String (auto-generated),
  customer: ObjectId (ref: Customer),
  items: [{
    product: ObjectId (ref: Product),
    quantity: Number,
    unitPrice: Number,
    totalPrice: Number
  }],
  totalAmount: Number,
  createdBy: ObjectId (ref: User)
}
\end{lstlisting}

\section*{APPENDIX C: INSTALLATION GUIDE}
\addcontentsline{toc}{section}{Appendix C: Installation Guide}

\subsection*{Prerequisites}
Node.js (v18+), MongoDB Atlas account, Git

\subsection*{Installation Steps}
\begin{lstlisting}[language=bash, caption=Installation Commands]
# Clone repository
git clone https://github.com/Princelad/stockify.git

# Backend setup
cd stockify/backend && npm install
cp .env.example .env  # Configure environment
npm run dev

# Frontend setup  
cd ../frontend && npm install
npm run dev
\end{lstlisting}

\subsection*{Environment Variables}
\begin{lstlisting}[caption=Required Environment Variables]
MONGO_URI=mongodb+srv://username:password@cluster.mongodb.net/db
JWT_SECRET=your-jwt-secret
GOOGLE_CLIENT_ID=your-google-client-id
GOOGLE_CLIENT_SECRET=your-google-client-secret
\end{lstlisting}

\section*{APPENDIX E: USER GUIDE}
\addcontentsline{toc}{section}{Appendix E: User Guide}

\subsection*{Getting Started}
\begin{enumerate}
\item Register account or login with Google
\item Set up business profile
\item Add product categories and suppliers
\item Import or add products manually
\item Start processing sales and tracking inventory
\end{enumerate}
